本文档提供所有自定义组件的使用示例和最佳实践。

## 一、基础UI组件

### 1. Card - 卡片组件

```tsx
import { Card } from '@/components/ui';

// 基础用法
<Card title="基础卡片">
  这是卡片内容
</Card>

// 毛玻璃效果
<Card title="毛玻璃卡片" glass shadow="xl">
  这是带毛玻璃效果的卡片
</Card>

// 可悬停卡片
<Card title="可悬停卡片" hoverable onClick={() => console.log('clicked')}>
  鼠标悬停时会有提升效果
</Card>

// 带额外内容的卡片
<Card
  title="带操作的卡片"
  extra={<Button type="link">更多</Button>}
>
  卡片内容
</Card>
```

### 2. Button - 按钮组件

```tsx
import { Button } from '@/components/ui';
import { PlusOutlined } from '@ant-design/icons';

// 不同类型
<Button type="primary">主要按钮</Button>
<Button type="secondary">次要按钮</Button>
<Button type="success">成功按钮</Button>
<Button type="danger">危险按钮</Button>
<Button type="ghost">幽灵按钮</Button>

// 不同尺寸
<Button size="sm">小按钮</Button>
<Button size="base">默认按钮</Button>
<Button size="lg">大按钮</Button>

// 带图标
<Button type="primary" icon={<PlusOutlined />}>
  新增
</Button>

// 加载状态
<Button type="primary" loading>
  加载中
</Button>

// 块级按钮
<Button type="primary" block>
  块级按钮
</Button>
```

### 3. Table - 表格组件

```tsx
import { Table } from '@/components/ui';

const columns = [
  {
    title: '姓名',
    dataIndex: 'name',
    key: 'name',
  },
  {
    title: '年龄',
    dataIndex: 'age',
    key: 'age',
    sorter: true,
  },
  {
    title: '操作',
    key: 'action',
    render: (_, record) => (
      <Button type="link">编辑</Button>
    ),
  },
];

const dataSource = [
  { key: '1', name: '张三', age: 32 },
  { key: '2', name: '李四', age: 28 },
];

// 基础用法
<Table columns={columns} dataSource={dataSource} />

// 毛玻璃效果 + 紧凑模式
<Table
  columns={columns}
  dataSource={dataSource}
  glass
  density="compact"
  striped
/>

// 带分页
<Table
  columns={columns}
  dataSource={dataSource}
  pagination={{
    current: 1,
    pageSize: 10,
    total: 100,
    onChange: (page, pageSize) => console.log(page, pageSize),
  }}
/>
```

### 4. Form - 表单组件

```tsx
import { Form, Button } from '@/components/ui';
import { Input, Select } from 'antd';

const [form] = Form.useForm();

<Form
  form={form}
  layout="vertical"
  initialValues={{ name: '', age: '' }}
  onFinish={(values) => console.log(values)}
>
  <Form.Item name="name" label="姓名" required>
    <Input placeholder="请输入姓名" />
  </Form.Item>

  <Form.Item name="age" label="年龄">
    <Input placeholder="请输入年龄" />
  </Form.Item>

  <Form.Item>
    <Button type="primary" htmlType="submit">
      提交
    </Button>
  </Form.Item>
</Form>
```

### 5. Modal - 模态框组件

```tsx
import { Modal, Button } from '@/components/ui';
import { useState } from 'react';

const [visible, setVisible] = useState(false);

<>
  <Button onClick={() => setVisible(true)}>打开模态框</Button>

  <Modal
    visible={visible}
    title="模态框标题"
    glass
    onCancel={() => setVisible(false)}
    onOk={() => {
      console.log('确认');
      setVisible(false);
    }}
  >
    这是模态框内容
  </Modal>
</>
```

### 6. Drawer - 抽屉组件

```tsx
import { Drawer, Button } from '@/components/ui';
import { useState } from 'react';

const [visible, setVisible] = useState(false);

<>
  <Button onClick={() => setVisible(true)}>打开抽屉</Button>

  <Drawer
    visible={visible}
    title="抽屉标题"
    placement="right"
    glass
    onClose={() => setVisible(false)}
  >
    这是抽屉内容
  </Drawer>
</>
```

## 四、完整页面示例

### 用户管理页面

```tsx
import React, { useState } from 'react';
import { PageContainer, SectionCard } from '@/components/layout';
import { FilterBar, ActionBar, StatusTag } from '@/components/business';
import { Table, Button, Modal } from '@/components/ui';
import { PlusOutlined, EditOutlined, DeleteOutlined } from '@ant-design/icons';

const UserManagementPage: React.FC = () => {
  const [selectedRowKeys, setSelectedRowKeys] = useState<React.Key[]>([]);

  const columns = [
    {
      title: '用户名',
      dataIndex: 'username',
      key: 'username',
    },
    {
      title: '姓名',
      dataIndex: 'name',
      key: 'name',
    },
    {
      title: '状态',
      dataIndex: 'status',
      key: 'status',
      render: (status: number) => (
        <StatusTag status={status === 1 ? 'active' : 'inactive'} />
      ),
    },
    {
      title: '操作',
      key: 'action',
      render: (_, record) => (
        <>
          <Button type="link" icon={<EditOutlined />}>编辑</Button>
          <Button type="link" danger icon={<DeleteOutlined />}>删除</Button>
        </>
      ),
    },
  ];

  const dataSource = [
    { key: '1', username: 'admin', name: '管理员', status: 1 },
    { key: '2', username: 'user1', name: '用户1', status: 1 },
    { key: '3', username: 'user2', name: '用户2', status: 0 },
  ];

  return (
    <PageContainer
      title="用户管理"
      subTitle="管理系统用户信息"
      breadcrumb={{
        items: [
          { title: '首页', path: '/' },
          { title: '系统管理' },
          { title: '用户管理' },
        ],
      }}
    >
      <SectionCard title="筛选条件">
        <FilterBar
          items={[
            {
              name: 'username',
              label: '用户名',
              type: 'input',
            },
            {
              name: 'name',
              label: '姓名',
              type: 'input',
            },
            {
              name: 'status',
              label: '状态',
              type: 'select',
              options: [
                { label: '启用', value: 1 },
                { label: '禁用', value: 0 },
              ],
            },
          ]}
          onSearch={(values) => console.log('搜索', values)}
          onReset={() => console.log('重置')}
        />
      </SectionCard>

      <SectionCard>
        <ActionBar
          actions={[
            {
              key: 'add',
              label: '新增用户',
              icon: <PlusOutlined />,
              type: 'primary',
              onClick: () => console.log('新增'),
            },
            {
              key: 'delete',
              label: '批量删除',
              icon: <DeleteOutlined />,
              danger: true,
              disabled: selectedRowKeys.length === 0,
              onClick: () => console.log('删除'),
            },
          ]}
          extra={
            selectedRowKeys.length > 0 && (
              <span>已选择 {selectedRowKeys.length} 项</span>
            )
          }
          align="space-between"
        />

        <Table
          columns={columns}
          dataSource={dataSource}
          glass
          density="compact"
          striped
          rowSelection={{
            selectedRowKeys,
            onChange: (keys) => setSelectedRowKeys(keys),
          }}
          pagination={{
            current: 1,
            pageSize: 10,
            total: 100,
          }}
        />
      </SectionCard>
    </PageContainer>
  );
};

export default UserManagementPage;
```

## 五、最佳实践

### 1. 组件组合

优先使用组件组合而不是创建新组件：

```tsx
// ✅ 好的做法
<PageContainer title="用户管理">
  <SectionCard title="筛选条件">
    <FilterBar items={filterItems} />
  </SectionCard>

  <SectionCard>
    <ActionBar actions={actions} />
    <Table columns={columns} dataSource={data} />
  </SectionCard>
</PageContainer>

// ❌ 不好的做法 - 创建过多的自定义组件
<CustomUserManagementPage />
```

### 2. 样式定制

使用className和style进行样式定制：

```tsx
// ✅ 好的做法
<Card
  className="custom-card"
  style={{ marginBottom: 24 }}
>
  内容
</Card>

// ❌ 不好的做法 - 直接修改组件源码
```

### 3. 性能优化

对于大数据量的表格，使用虚拟滚动：

```tsx
<Table
  columns={columns}
  dataSource={largeDataSource}
  scroll={{ y: 500 }}
  virtual
/>
```

### 4. 响应式设计

使用响应式布局：

```tsx
<FilterBar
  items={filterItems}
  collapsible
  defaultRows={1}  // 默认只显示1行，其余折叠
/>
```
