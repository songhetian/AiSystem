## 📱 响应式设计优化

### 1. 基础用法

#### 在表格页面中使用

```typescript
import { useResponsive } from '@/hooks/useResponsive';

export default function MyTablePage() {
  const { isMobile, isTablet, isDesktop, getTableConfig } = useResponsive();

  return (
```

**效果**:

- 移动端：每页显示 5 条，隐藏部分功能

### 3. 响应式列配置

```typescript
import { useResponsiveColumns } from '@/hooks/useResponsive';

export default function MyTablePage() {
  const { getVisibleColumns } = useResponsiveColumns();

  const allColumns = [
    {
      title: 'ID',
      dataIndex: 'id',
      priority: 1  // 最高优先级，移动端也显示
    },
    {
      title: '名称',
      dataIndex: 'name',
      priority: 1
    },
    {
      title: '描述',
      dataIndex: 'desc',
      priority: 2  // 中等优先级，平板及以上显示
    },
    {
      title: '创建时间',
      dataIndex: 'createdAt',
      priority: 3  // 低优先级，仅桌面显示
    },
  ];

  // 根据屏幕大小自动过滤列
  const columns = getVisibleColumns(allColumns);

  return <ProTable columns={columns} />;
}
```

**优先级规则**:

- `priority: 1` - 所有设备都显示
- `priority: 2` - 平板及以上显示
- `priority: 3` - 仅桌面显示
- 无 `priority` - 等同于 `priority: 1`

## 🚀 数据预加载优化

### 1. 基础用法 - 鼠标悬停预加载

```typescript
import { useQueryClient } from '@tanstack/react-query';
import { prefetchData } from '@/utils/data-prefetch';

export default function EmployeeList() {
  const queryClient = useQueryClient();

  const handleRowHover = (record: any) => {
    // 鼠标悬停时预加载详情数据
    prefetchData({
      queryClient,
      queryKey: ['employee-detail', record.id],
      queryFn: () => fetchEmployeeDetail(record.id),
    });
  };

  return (
    <Table
      dataSource={employees}
      onRow={(record) => ({
        onMouseEnter: () => handleRowHover(record),
      })}
    />
  );
}
```

**效果**: 用户点击查看详情时，数据已经预加载完成，瞬间显示！

### 3. 批量预加载

```typescript
import { useQueryClient } from '@tanstack/react-query';

export default function Dashboard() {
  const queryClient = useQueryClient();

  useEffect(() => {
    // 页面加载时批量预加载相关数据
    prefetchBatch([
      {
        queryClient,
        queryKey: ['departments'],
        queryFn: () => fetchDepartments(),
      },
      {
        queryClient,
        queryKey: ['positions'],
        queryFn: () => fetchPositions(),
      },
      {
        queryClient,
        queryKey: ['recent-activities'],
        queryFn: () => fetchRecentActivities(),
      },
    ]);
  }, [queryClient]);

  return <div>Dashboard Content</div>;
}
```

### 6. 分页预加载

```typescript
import { useQueryClient } from '@tanstack/react-query';
import { prefetchData } from '@/utils/data-prefetch';

export default function PaginatedList() {
  const queryClient = useQueryClient();
  const [page, setPage] = useState(1);

  const { data } = useQuery({
    queryKey: ['list', page],
    queryFn: () => fetchList(page),
  });

  useEffect(() => {
    // 预加载下一页数据
    prefetchData({
      queryClient,
      queryKey: ['list', page + 1],
      queryFn: () => fetchList(page + 1),
    });
  }, [page, queryClient]);

  return (
    <Table
      dataSource={data}
      pagination={{
        current: page,
        onChange: setPage,
      }}
    />
  );
}
```

## ⚠️ 注意事项

### 响应式优化

1. 不要过度隐藏功能，确保核心功能在移动端可用
2. 测试不同设备的显示效果
3. 考虑横屏和竖屏的差异

### 数据预加载

1. 只预加载用户可能访问的数据
2. 避免预加载大量数据造成浪费
3. 设置合理的缓存时间（默认 5 分钟）
4. 预加载失败不应影响用户体验
